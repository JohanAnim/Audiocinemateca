rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- 1. FUNCIONES DE SEGURIDAD Y ROLES ---
    
    function isSuperAdmin() {
      return request.auth != null && request.auth.token.email == "gutierrezjohanantonio@gmail.com";
    }

    function isAdmin() {
      return isSuperAdmin() || (request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin");
    }

    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    function isBanned(userId) {
      return exists(/databases/$(database)/documents/banned_users/$(userId));
    }

    // --- 2. PERFILES Y DATOS DE USUARIO EN LA NUBE ---
    
    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if isOwner(userId) || isAdmin();
    }

    match /users/{userId}/{document=**} {
      allow read: if isOwner(userId) || isAdmin();
      allow write: if isOwner(userId) || isAdmin();
    }

    match /{path=**}/history/{historyId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    // --- 3. CHAT GLOBAL Y CONFIGURACIÓN ---
    
    match /global_chat/{messageId} {
      // Todos los usuarios autenticados pueden leer el chat
      allow read: if request.auth != null;
      
      // Permitir enviar mensajes si el usuario está logueado, es el remitente y NO está baneado
      allow create: if request.auth != null && 
        request.resource.data.senderId == request.auth.uid &&
        !isBanned(request.auth.uid);
      
      // Permite editar el texto de sus propios mensajes o actualizar reacciones
      allow update: if isAdmin() || (
        request.auth != null && !isBanned(request.auth.uid)
      );
      
      // Solo los Admins pueden eliminar mensajes
      allow delete: if isAdmin();
    }

    match /config/chat_global {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // --- 4. TRANSMISIONES JAM EN VIVO (SALAS DE ESCUCHA) ---

    match /global_chat_jams/{jamId} {
      allow read: if request.auth != null;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        request.auth != null && !isBanned(request.auth.uid)
      );
    }

    // --- 5. PRESENCIA DE USUARIOS Y LISTA EN VIVO ---
    
    match /presence/{userId} {
      allow read: if request.auth != null;
      allow write: if isOwner(userId);
    }

    // --- 6. RANKING TOP 10 SEMANAL Y CACHÉ GLOBAL ---
    
    match /global_rankings/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /ranking_cache/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 7. BANNER DESTACADO Y CONFIGURACIÓN DE LA APP ---
    
    match /app_config/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // --- 8. COMENTARIOS, REACCIONES Y REPORTES ---
    
    match /{path=**}/comments/{commentId} {
      allow read: if true;
      allow write: if request.auth != null && !isBanned(request.auth.uid);
    }
    
    match /content_comments/{document=**} {
      allow read: if true;
      allow write: if request.auth != null && !isBanned(request.auth.uid);
    }

    match /comment_reports/{reportId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && !isBanned(request.auth.uid);
      allow update, delete: if isAdmin();
    }

    // --- 9. ANUNCIOS DE LA COMUNIDAD Y REACCIONES ---
    
    match /anuncios/{announcementId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        request.auth != null && !isBanned(request.auth.uid)
      );
    }

    // --- 10. VOTOS (ME GUSTA / NO ME GUSTA) Y SUBCOLECCIONES ---
    
    match /content_votes/{document=**} {
      allow read: if true;
      allow write: if request.auth != null && !isBanned(request.auth.uid);
    }

    match /{path=**}/user_votes/{voteId} {
      allow read: if true;
      allow write: if request.auth != null && !isBanned(request.auth.uid);
    }

    // --- 11. CALIFICACIONES, ESTRELLAS Y SUBCOLECCIONES ---
    
    match /content_ratings/{document=**} {
      allow read: if true;
      allow write: if request.auth != null && !isBanned(request.auth.uid);
    }

    match /{path=**}/user_ratings/{ratingId} {
      allow read: if true;
      allow write: if request.auth != null && !isBanned(request.auth.uid);
    }

    // --- 12. ESTADÍSTICAS Y REPRODUCCIONES DE COMUNIDAD ---
    
    match /content_stats/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 13. USUARIOS SANCIONADOS / BANEADOS ---
    
    match /banned_users/{userId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // --- 14. RED DE SEGURIDAD Y ADMINISTRACIÓN ---
    
    match /{document=**} {
      allow read, write: if isAdmin();
    }
  }
}
