rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- 1. FUNCIONES DE PODER ---
    
    function isSuperAdmin() {
      return request.auth != null && request.auth.token.email == "gutierrezjohanantonio@gmail.com";
    }

    function isAdmin() {
      return isSuperAdmin() || (request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin");
    }

    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    // --- 2. PERFILES Y DATOS DE USUARIO ---
    
    match /users/{userId}/{document=**} {
      allow read, write: if isOwner(userId) || isAdmin();
    }

    // --- 3. CHAT GLOBAL (NUEVO) ---
    match /global_chat/{messageId} {
      // Todos los logueados pueden leer el chat
      allow read: if request.auth != null;
      // Solo logueados pueden escribir sus propios mensajes
      allow create: if request.auth != null && request.resource.data.senderId == request.auth.uid;
      // NADIE puede borrar o editar mensajes excepto los Admins
      allow update, delete: if isAdmin();
    }

    match /config/chat_global {
      // Todos pueden leer si el chat está abierto o cerrado
      allow read: if request.auth != null;
      // Solo los Admins pueden cambiar el estado
      allow write: if isAdmin();
    }

    // --- 4. COMENTARIOS ---
    match /{path=**}/comments/{commentId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /content_comments/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 5. ANUNCIOS ---
    match /anuncios/{announcementId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        request.auth != null && 
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['reactions'])
      );
    }

    // --- 6. VOTOS (LIKES/DISLIKES) ---
    match /content_votes/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 7. CALIFICACIONES (ESTRELLAS) ---
    match /content_ratings/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 8. ESTADÍSTICAS Y VISTAS ---
    match /content_stats/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 9. RED DE SEGURIDAD ---
    match /{document=**} {
      allow read, write: if isAdmin();
    }
  }
}
