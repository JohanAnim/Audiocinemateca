rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- 1. FUNCIONES DE PODER ---
    
    function isSuperAdmin() {
      return request.auth != null && request.auth.token.email == "gutierrezjohanantonio@gmail.com";
    }

    function isAdmin() {
      return isSuperAdmin() || (request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin");
    }

    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    // --- 2. PERFILES Y DATOS DE USUARIO ---
    
    match /users/{userId}/{document=**} {
      allow read, write: if isOwner(userId) || isAdmin();
    }

    // --- 3. COMENTARIOS ---
    // Regla para subcolecciones y soporte para Collection Group Queries
    match /{path=**}/comments/{commentId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /content_comments/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 4. ANUNCIOS (NUEVO) ---
    match /anuncios/{announcementId} {
      allow read: if true;
      // Solo admins pueden crear o borrar anuncios
      allow create, delete: if isAdmin();
      // Usuarios pueden actualizar SOLO el campo de reacciones
      allow update: if isAdmin() || (
        request.auth != null && 
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['reactions'])
      );
    }

    // --- 5. VOTOS (LIKES/DISLIKES) ---
    match /content_votes/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 6. CALIFICACIONES (ESTRELLAS) ---
    match /content_ratings/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 7. ESTADÍSTICAS Y VISTAS ---
    match /content_stats/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // --- 8. RED DE SEGURIDAD ---
    match /{document=**} {
      allow read, write: if isAdmin();
    }
  }
}
